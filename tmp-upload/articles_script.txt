<style>
#bz-home .bz-filters{display:flex;flex-wrap:wrap;gap:8px;margin:0 0 28px}
#bz-home .bz-filter{background:transparent;border:1px solid rgba(238,242,245,.22);color:#EEF2F5;border-radius:999px;padding:8px 15px;font:500 14px/1.2 Inter,system-ui,sans-serif;cursor:pointer}
#bz-home .bz-filter:hover{border-color:#8FDCA6}
#bz-home .bz-filter.on{background:#8FDCA6;border-color:#8FDCA6;color:#092142}
#bz-home .bz-hub{color:#8FDCA6}
</style>
<script>
(function () {
  // Article cards from the WordPress REST API, filtered by topic hub (WordPress categories).
  // Misinformation-archive posts stay live at their URLs but are left out of this list.
  var ARCHIVE = 39;
  var HUBS = [
    { id: 0, name: "All", slug: "" },
    { id: 33, name: "Energy and the grid", slug: "energy-and-the-grid" },
    { id: 34, name: "Methane", slug: "methane" },
    { id: 35, name: "Emissions and data", slug: "emissions-and-data" },
    { id: 36, name: "Money and human rights", slug: "money-and-human-rights" },
    { id: 37, name: "Adoption and investors", slug: "adoption-and-investors" },
    { id: 38, name: "Reflections", slug: "reflections" }
  ];
  var NAME = {}; HUBS.forEach(function (h) { if (h.id) NAME[h.id] = h.name; });
  var UP = "/wp-content/uploads/2026/11/";
  var DESIGNED = {
    "banking-on-biometrics-is-fraud-prevention-the-real-goal": UP + "article-banking-on-biometrics.png",
    "the-changing-narrative-on-bitcoin-mining": UP + "article-new-data-backed-story.png",
    "fact-and-fiction-in-bitcoin-mining-reporting": UP + "article-fact-and-fiction.png",
    "bitcoin-coverage-as-a-litmus-test-for-media-reliability": UP + "article-litmus-test.png"
  };
  var FALLBACK = UP + "batcoinz-banner-five-paths-scaled.png";
  var MONTHS = ["Jan","Feb","Mar","Apr","May","Jun","Jul","Aug","Sep","Oct","Nov","Dec"];
  document.getElementById("bz-year").textContent = new Date().getFullYear();
  function fmt(d) { var t = new Date(d); return t.getDate() + " " + MONTHS[t.getMonth()] + " " + t.getFullYear(); }
  function firstImage(html) { var m = /<img[^>]+src="([^"]+)"/i.exec(html || ""); return m ? m[1] : ""; }
  var grid = document.getElementById("bz-grid"), btn = document.getElementById("bz-more-btn"), count = document.getElementById("bz-count");
  var bar = document.getElementById("bz-filters");
  var page = 1, busy = false, hub = HUBS[0], seq = 0;
  HUBS.forEach(function (h) {
    var b = document.createElement("button"); b.type = "button"; b.className = "bz-filter"; b.textContent = h.name; b.dataset.slug = h.slug;
    b.addEventListener("click", function () { select(h, true); });
    bar.appendChild(b);
  });
  function select(h, push) {
    hub = h; page = 1; grid.innerHTML = "";
    [].forEach.call(bar.children, function (b) { var on = b.dataset.slug === h.slug; b.classList.toggle("on", on); b.setAttribute("aria-pressed", on); });
    if (push) { try { history.replaceState(null, "", h.slug ? "#" + h.slug : location.pathname); } catch (e) {} }
    load();
  }
  function load() {
    var my = ++seq; busy = true; btn.disabled = true; btn.textContent = "Loading…";
    var filter = hub.id ? "&categories=" + hub.id : "&categories_exclude=" + ARCHIVE;
    fetch("/wp-json/wp/v2/posts?per_page=12&page=" + page + filter + "&_embed=wp:featuredmedia&_fields=slug,date,link,title,content,categories,_links,_embedded")
      .then(function (r) {
        if (my !== seq) return [];
        var total = r.headers.get("X-WP-Total"), pages = +r.headers.get("X-WP-TotalPages") || 1;
        btn.hidden = page >= pages; btn.disabled = false; btn.textContent = "Show more articles"; busy = false;
        if (total) count.textContent = total + (total === "1" ? " article" : " articles") + (hub.id ? " in " + hub.name : "");
        return r.json();
      })
      .then(function (posts) {
        if (my !== seq) return;
        posts.forEach(function (p) {
          var fm = p._embedded && p._embedded["wp:featuredmedia"] && p._embedded["wp:featuredmedia"][0];
          var img = (fm && fm.source_url) || DESIGNED[p.slug] || firstImage(p.content.rendered) || FALLBACK;
          var a = document.createElement("a"); a.className = "bz-card"; a.href = p.link;
          var im = document.createElement("img"); im.src = img; im.alt = ""; im.loading = "lazy";
          im.onerror = function () { if (im.src.indexOf(FALLBACK) < 0) im.src = FALLBACK; };
          var body = document.createElement("div"); body.className = "bz-card-body";
          var d = document.createElement("span"); d.className = "bz-date"; d.textContent = fmt(p.date);
          var hn = (p.categories || []).map(function (c) { return NAME[c]; }).filter(Boolean)[0];
          if (hn && !hub.id) { var s = document.createElement("span"); s.className = "bz-hub"; s.textContent = " · " + hn; d.appendChild(s); }
          var h = document.createElement("h3"); h.innerHTML = p.title.rendered;
          body.appendChild(d); body.appendChild(h); a.appendChild(im); a.appendChild(body); grid.appendChild(a);
        });
      })
      .catch(function () { busy = false; btn.disabled = false; btn.textContent = "Show more articles"; });
  }
  btn.addEventListener("click", function () { if (!busy) { page++; load(); } });
  var start = HUBS.filter(function (h) { return h.slug && "#" + h.slug === location.hash; })[0] || HUBS[0];
  select(start, false);
})();
</script>
